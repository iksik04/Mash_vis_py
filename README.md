Пресеты имеют следующую структуру
Ref_img_context.json
{
  "version": 1,                         //версия для контроля совместимости, version
  "reference_image": "ref.jpg",         //исходное brg изображение, захваченное камерой, ref_img
  "reference_bin_image": "ref_bin.jpg", //бинаризованное изображение, ref_bin_img
  "reference_center": [float, float],   //координаты центра бинаризованного изображения, [rcx, rcy]
  "reference_contour": [...],           //массив точек найденного контура, ref_ext_contour
  "binarization_params": {
        "reference_low_threshold": int  //BINARYSATION_LOW_THRESHOLD для получения ref_bin_img
        "reference_high_threshold": int //BINARYSATION_HIGH_THRES HOLD для получения ref_bin_img
    }
}

ROI.json
